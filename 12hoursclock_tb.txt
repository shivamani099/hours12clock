module tb;
  wire [5:0] sec,min;
  reg mode,set,inc,dec;
  wire [3:0] hrs12;
  wire am_pm;
  reg clk=0,rst;

  always #5 clk = ~clk;

  digitalclock12h dut (clk,rst,inc,dec,mode,set,sec,min,hrs12,am_pm);

  initial begin
    rst = 1;@(posedge clk) rst = 0;
    @(posedge clk);
    mode=1;
    @(posedge clk);
    mode=1;
    inc=1;
    dec=1;
    @(posedge clk);
  end
  
  always @(posedge clk) begin
      if (am_pm == 0)
        $display("TIME = %02d:%02d:%02d AM",
                  hrs12, min, sec);
      else
        $display("TIME = %02d:%02d:%02d PM",
                  hrs12, min, sec);
  end

  initial begin
    repeat(1000) @(posedge clk);
    $finish;
  end
 
endmodule
